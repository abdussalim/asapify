    python - <<'EOF'
import io, os, httpx, pandas as pd, datetime as dt
K, B = os.environ["MAP_KEY"], os.environ["BACKTEST_BBOX"]
url = "https://firms.modaps.eosdis.nasa.gov/api/area/csv/{}/{}/{}/5/{}"

EOF

Prasyarat

Keluaran

  • data/local/raw/himawari/ (4 file), data/local/raw/gk2a/ (2 file)

Langkah

  1. Unduh Himawari-9 B07 + B14 segmen S05 dan S06 untuk 25 Sep 2023 15.00 UTC, catat ukurannya.
    python - <<'EOF'
import s3fs
fs = s3fs.S3FileSystem(anon=True)
p = "noaa-himawari9/AHI-L1b-FLDK/2023/09/25/1500"
files = [f for f in fs.ls(p) if any(b in f for b in ("_B07_", "_B14_")) and ("S0510" in f or "S0610" in f)]
for f in files:
    print(f.split("/")[-1], round(fs.size(f)/1e6, 1), "MB"); fs.get(f, "data/local/raw/himawari/")
EOF
  2. Unduh GK2A SW038 + IR105 untuk waktu yang sama.
    python - <<'EOF'
import s3fs
fs = s3fs.S3FileSystem(anon=True)
p = "noaa-gk2a-pds/AMI/L1B/FD/202309/25/15"
for ch in ("sw038", "ir105"):
    f = f"{p}/gk2a_ami_le1b_{ch}_fd020ge_202309251500.nc"
    print(ch, round(fs.size(f)/1e6, 1), "MB"); fs.get(f, "data/local/raw/gk2a/")
EOF
  3. Baca keduanya ke grid gabungan dan bandingkan (satu area di Sumatra, satu di Kalimantan).
    python - <<'EOF'
import glob, numpy as np
from satpy import Scene
from jobs.lib.region import AREA
def grid(reader, files, ch):
    s = Scene(reader=reader, filenames=files); s.load([ch], calibration="brightness_temperature")
    return s.resample(AREA, resampler="nearest", radius_of_influence=5000)[ch].values
h = grid("ahi_hsd", sorted(glob.glob("data/local/raw/himawari/*B07*")), "B07")
g = grid("ami_l1b", glob.glob("data/local/raw/gk2a/*sw038*"), "SW038")
for n, a in (("Himawari B07", h), ("GK2A SW038", g)):
    print(n, "NaN:", round(float(np.isnan(a).mean()), 4), "min/max K:", np.nanmin(a), np.nanmax(a))
m = ~np.isnan(h) & ~np.isnan(g)
print("korelasi:", round(float(np.corrcoef(h[m], g[m])[0, 1]), 3))
EOF
  4. Ukur latensi kedua bucket 3 kali (sore, malam, dini hari WIB): selisih waktu scan terbaru dan LastModified. Tambahkan ke data/local/latency.csv.
    python - <<'EOF'
import s3fs, datetime as dt
fs = s3fs.S3FileSystem(anon=True); now = dt.datetime.utcnow()
h = sorted(fs.ls(f"noaa-himawari9/AHI-L1b-FLDK/{now:%Y/%m/%d}"))[-1]
hf = [x for x in fs.ls(h) if "_B07_" in x and "S0510" in x][0]
hs = dt.datetime.strptime(h[-15:], "%Y/%m/%d/%H%M")
g = sorted(fs.ls(f"noaa-gk2a-pds/AMI/L1B/FD/{now:%Y%m/%d}"))[-1]
gf = sorted(x for x in fs.ls(g) if "sw038" in x)[-1]
gs = dt.datetime.strptime(gf[-15:-3], "%Y%m%d%H%M")
for n, s, f in (("himawari", hs, hf), ("gk2a", gs, gf)):
    lm = fs.info(f)["LastModified"].replace(tzinfo=None)
    print(f"{n},{s:%Y-%m-%dT%H:%MZ},{(lm - s).total_seconds()/60:.1f}")
EOF

Cek selesai

Kedua satelit: NaN < 0.01, min/max 200–340 K, korelasi ≥ 0,8. Bila NaN Himawari ≥ 0,01: tambah segmen S04/S07 di himawari.py.

Bila satpy gagal membaca .bz2: jalankan bunzip2 -k dulu dan catat di WORKLOG ⚠️.

valid = [k for k, v in p["TS"].items() if v != -999]
print("sumber:", d["header"]["sources"], "jam valid terakhir:", valid[-1], "TS:", p["TS"][valid[-1]], "T2M:", p["T2M"][valid[-1]])
